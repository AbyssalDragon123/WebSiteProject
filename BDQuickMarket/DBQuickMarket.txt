--------------------------------------------------------------------------------CREAR BASE DE DATOS USUARIO Y TABLESPACE
-- 1) TABLESPACE 
--------------------------------------------------------------------------------
CREATE TABLESPACE QUICKMARKET_TBS
  DATAFILE 'quickmarket01.dbf'
  SIZE 50M
  AUTOEXTEND ON NEXT 10M
  MAXSIZE UNLIMITED;

--------------------------------------------------------------------------------
-- 2) USUARIO NUEVO
--------------------------------------------------------------------------------
CREATE USER QUICKMARKET_USER IDENTIFIED BY QuickMkt#2025
  DEFAULT TABLESPACE QUICKMARKET_TBS
  TEMPORARY TABLESPACE TEMP
  QUOTA UNLIMITED ON QUICKMARKET_TBS;

-- Permisos básicos
GRANT CONNECT, RESOURCE TO QUICKMARKET_USER;

-----------------------------------------------------------------------------
-- CREAR LA TABLA DE DATOS 
--------------------------------------------------------------------------------
-- 4) TABLAS DEL ESQUEMA "QuickMarket"
--------------------------------------------------------------------------------

-- 4.1 USUARIOS
CREATE TABLE USUARIOS (
  ID_USUARIO           NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  USUARIO              VARCHAR2(50)  NOT NULL UNIQUE,
  EMAIL                VARCHAR2(150) NOT NULL UNIQUE,
  PASSWORD             VARCHAR2(255) NOT NULL,
  ROL                  VARCHAR2(20)  NOT NULL,
  ESTADO               VARCHAR2(20)  DEFAULT 'activo' NOT NULL,
  RESET_TOKEN          VARCHAR2(255),
  RESET_TOKEN_EXPIRA   TIMESTAMP,
  CREADO_EN            TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  ACTUALIZADO_EN       TIMESTAMP
) TABLESPACE QUICKMARKET_TBS;

COMMENT ON COLUMN USUARIOS.PASSWORD IS 'Guardar SOLO el hash (nunca en texto plano).';

-- 4.2 CLIENTES
CREATE TABLE CLIENTES (
  ID_CLIENTE     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  ID_USUARIO     NUMBER NOT NULL,
  NOMBRE         VARCHAR2(150) NOT NULL,
  TELEFONO       VARCHAR2(30),
  DIRECCION      VARCHAR2(200),
  DEPARTAMENTO   VARCHAR2(80),
  MUNICIPIO      VARCHAR2(80),
  REFERENCIA     VARCHAR2(200),
  CONSTRAINT FK_CLIENTE_USUARIO
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIOS(ID_USUARIO)
    ON DELETE CASCADE
) TABLESPACE QUICKMARKET_TBS;

CREATE INDEX IX_CLIENTES_USUARIO ON CLIENTES(ID_USUARIO);

-- 4.3 CATEGORIAS
CREATE TABLE CATEGORIAS (
  ID_CATEGORIA        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  NOMBRE              VARCHAR2(100) NOT NULL,
  DESCRIPCION         VARCHAR2(255),
  ID_CATEGORIA_PADRE  NUMBER NULL,
  CONSTRAINT FK_CAT_PADRE
    FOREIGN KEY (ID_CATEGORIA_PADRE) REFERENCES CATEGORIAS(ID_CATEGORIA)
    ON DELETE SET NULL,
  CONSTRAINT CHK_CAT_NO_SELF CHECK (ID_CATEGORIA_PADRE IS NULL OR ID_CATEGORIA_PADRE <> ID_CATEGORIA)
) TABLESPACE QUICKMARKET_TBS;

CREATE INDEX IX_CATEGORIAS_PADRE ON CATEGORIAS(ID_CATEGORIA_PADRE);

-- 4.4 PRODUCTOS
CREATE TABLE PRODUCTOS (
  ID_PRODUCTO       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  NOMBRE            VARCHAR2(120)  NOT NULL,
  CODIGO_BARRAS     VARCHAR2(50),
  PRECIO_UNITARIO   NUMBER(12,2)   NOT NULL,
  STOCK             NUMBER(12,3)   DEFAULT 0 NOT NULL,
  ID_CATEGORIA      NUMBER NULL,
  CONSTRAINT FK_PRODUCTO_CATEGORIA
    FOREIGN KEY (ID_CATEGORIA) REFERENCES CATEGORIAS(ID_CATEGORIA)
    ON DELETE SET NULL,
  CONSTRAINT CHK_PRECIO_POSITIVO CHECK (PRECIO_UNITARIO >= 0),
  CONSTRAINT CHK_STOCK_NO_NEG    CHECK (STOCK >= 0),
  CONSTRAINT UQ_PRODUCTO_CODBARR UNIQUE (CODIGO_BARRAS)
) TABLESPACE QUICKMARKET_TBS;

CREATE INDEX IX_PRODUCTOS_CATEGORIA ON PRODUCTOS(ID_CATEGORIA);

-- 4.5 VENTAS
CREATE TABLE VENTAS (
  ID_VENTA          NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  FECHA             DATE DEFAULT SYSDATE NOT NULL,
  ID_CLIENTE        NUMBER,
  TOTAL_BRUTO       NUMBER(12,2) NOT NULL,
  TOTAL_IMPUESTOS   NUMBER(12,2) NOT NULL,
  TOTAL_NETO        NUMBER(12,2) NOT NULL,
  ESTADO            VARCHAR2(20) NOT NULL,
  ID_USUARIO        NUMBER, -- vendedor/cajero
  CONSTRAINT FK_VENTA_CLIENTE
    FOREIGN KEY (ID_CLIENTE) REFERENCES CLIENTES(ID_CLIENTE)
    ON DELETE SET NULL,
  CONSTRAINT FK_VENTA_USUARIO
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIOS(ID_USUARIO)
    ON DELETE SET NULL,
  CONSTRAINT CHK_TOTALES_NO_NEG CHECK (TOTAL_BRUTO >= 0 AND TOTAL_IMPUESTOS >= 0 AND TOTAL_NETO >= 0)
) TABLESPACE QUICKMARKET_TBS;

CREATE INDEX IX_VENTAS_CLIENTE ON VENTAS(ID_CLIENTE);

-- 4.6 DETALLE_VENTAS
CREATE TABLE DETALLE_VENTAS (
  ID_DETALLE        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  ID_VENTA          NUMBER NOT NULL,
  ID_PRODUCTO       NUMBER NOT NULL,
  CANTIDAD          NUMBER(12,3) NOT NULL,
  PRECIO_UNITARIO   NUMBER(12,2) NOT NULL,
  SUBTOTAL          GENERATED ALWAYS AS (CANTIDAD * PRECIO_UNITARIO) VIRTUAL,
  CONSTRAINT FK_DET_VENTA
    FOREIGN KEY (ID_VENTA) REFERENCES VENTAS(ID_VENTA)
    ON DELETE CASCADE,
  CONSTRAINT FK_DET_PRODUCTO
    FOREIGN KEY (ID_PRODUCTO) REFERENCES PRODUCTOS(ID_PRODUCTO),
  CONSTRAINT CHK_DET_CANT_POS CHECK (CANTIDAD > 0),
  CONSTRAINT CHK_DET_PREC_POS CHECK (PRECIO_UNITARIO >= 0),
  CONSTRAINT UQ_DET_VENTA_PRODUCTO UNIQUE (ID_VENTA, ID_PRODUCTO)
) TABLESPACE QUICKMARKET_TBS;

CREATE INDEX IX_DETALLE_VENTAS_VENTA    ON DETALLE_VENTAS(ID_VENTA);
CREATE INDEX IX_DETALLE_VENTAS_PRODUCTO ON DETALLE_VENTAS(ID_PRODUCTO);
